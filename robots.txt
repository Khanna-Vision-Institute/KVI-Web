# Updated SEO + AI robots.txt

# Named crawler groups were removed on purpose. A named User-agent block
# does not inherit the * group's Disallow rules, so Googlebot, Bingbot,
# and the AI crawlers listed here previously ignored the utm_ and
# refineParams blocks. With only *, those bots stay allowed and pick up
# the same disallows. The empty "Private" comment section was removed.

User-agent: *
Allow: /

# Block duplicate / tracking URLs
Disallow: /*?utm_
Disallow: /*&utm_
Disallow: /*?refineParams=
